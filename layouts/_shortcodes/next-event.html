{{/* Hero badge linking to the next upcoming event (by event_end >= now). Falls back to a link to /events-page/ when no event is upcoming. Usage: {{< next-event >}} */}} {{- $tz := "Europe/London" -}} {{- $next := dict -}} {{- with site.GetPage "/events" -}} {{- range .RegularPages -}} {{- $s := time.AsTime .Params.event_start $tz -}} {{- $e := time.AsTime (.Params.event_end | default .Params.event_start) $tz -}} {{- if $e.After now -}} {{- if or (not $next.page) ($s.Before $next.start) }}{{ $next = dict "page" . "start" $s }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- $href := "/events-page/" -}} {{- $text := "See upcoming events" -}} {{- with $next.page }}{{ $href = .RelPermalink }}{{ $text = printf "Next event: %s, %s" .Title ($next.start.Format "2 January 2006") }}{{ end -}} {{ $text }}
